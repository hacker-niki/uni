-- Проверка связанных записей до удаления
mysql> SELECT COUNT(*) FROM test_sessions WHERE user_id = 5;
+----------+
| COUNT(*) |
+----------+
| 12 |
+----------+
 
mysql> SELECT COUNT(*) FROM user_answers ua
-> JOIN test_sessions ts ON ua.test_session_id = ts.id
-> WHERE ts.user_id = 5;
+----------+
| COUNT(*) |
+----------+
| 180 |
+----------+
 
-- Удаление пользователя (триггер каскадное удаление)
mysql> DELETE FROM users WHERE id = 5;
Query OK, 1 row affected (0.03 sec)
 
-- Проверка каскадного удаления
mysql> SELECT COUNT(*) FROM test_sessions WHERE user_id = 5;
+----------+
| COUNT(*) |
+----------+
| 0 | ← Все сессии удалены
+----------+
 
mysql> SELECT COUNT(*) FROM user_answers ua
-> LEFT JOIN test_sessions ts ON ua.test_session_id = ts.id
-> WHERE ts.id IS NULL;
+----------+
| COUNT(*) |
+----------+
| 0 | ← Осиротевших записей нет
+----------+
 
✓ Каскадное удаление работает корректно
